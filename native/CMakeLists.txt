cmake_minimum_required(VERSION 3.24)
project(voice_router_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(whisper GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG 2eeeba56e9edd762b4b38467bab96c2517163158)
FetchContent_MakeAvailable(whisper)
FetchContent_Declare(fasttext GIT_REPOSITORY https://github.com/facebookresearch/fastText.git GIT_TAG 5b5943c118b0ec5fb9cd8d20587de2b2d3966dfe)
FetchContent_GetProperties(fasttext)
if(NOT fasttext_POPULATED)
  FetchContent_Populate(fasttext)
endif()
file(GLOB FASTTEXT_SOURCES CONFIGURE_DEPENDS "${fasttext_SOURCE_DIR}/src/*.cc")
list(REMOVE_ITEM FASTTEXT_SOURCES "${fasttext_SOURCE_DIR}/src/main.cc")
add_library(router_fasttext STATIC ${FASTTEXT_SOURCES})
target_include_directories(router_fasttext PUBLIC "${fasttext_SOURCE_DIR}/src")
if(NOT MSVC)
  # v0.9.2 relies on transitive exception headers; libc++ requires an explicit include.
  target_compile_options(router_fasttext PUBLIC -include exception)
endif()
find_package(Threads REQUIRED)
target_link_libraries(router_fasttext PUBLIC Threads::Threads)
add_executable(fasttext "${fasttext_SOURCE_DIR}/src/main.cc")
target_link_libraries(fasttext PRIVATE router_fasttext)
add_library(voice_router_native SHARED bridge.cpp)
set_target_properties(voice_router_native PROPERTIES PREFIX "")
target_link_libraries(voice_router_native PRIVATE whisper router_fasttext)
if(MINGW)
  target_link_options(voice_router_native PRIVATE -static-libstdc++)
  target_link_options(fasttext PRIVATE -static-libstdc++)
endif()
